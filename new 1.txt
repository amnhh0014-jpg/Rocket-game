    import * as THREE from 'three';

    /* =========================================================================
       1. PROCEDURAL SOUND SYNTHESIZER (ZERO ASSET DEPENDENCY)
       ========================================================================= */
    class SoundFX {
      constructor() {
        this.ctx = null;
      }
      init() {
        if (!this.ctx) {
          const AudioContext = window.AudioContext || window.webkitAudioContext;
          this.ctx = new AudioContext();
        }
        if (this.ctx && this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
      }
      jump() {
        if (!this.ctx) return;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'triangle';
        osc.frequency.setValueAtTime(140, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(420, this.ctx.currentTime + 0.15);
        gain.gain.setValueAtTime(0.3, this.ctx.currentTime);
        gain.gain.linearRampToValueAtTime(0, this.ctx.currentTime + 0.15);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.15);
      }
      dodge() {
        if (!this.ctx) return;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(300, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(80, this.ctx.currentTime + 0.25);
        gain.gain.setValueAtTime(0.35, this.ctx.currentTime);
        gain.gain.linearRampToValueAtTime(0, this.ctx.currentTime + 0.25);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.25);
      }
      hit() {
        if (!this.ctx) return;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(180, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(40, this.ctx.currentTime + 0.18);
        gain.gain.setValueAtTime(0.5, this.ctx.currentTime);
        gain.gain.linearRampToValueAtTime(0, this.ctx.currentTime + 0.18);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.18);
      }
      goalExplosion() {
        if (!this.ctx) return;
        // White noise burst + low sub drop
        const bufferSize = this.ctx.sampleRate * 1.5;
        const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const data = buffer.getChannelData(0);
        for (let i = 0; i < bufferSize; i++) {
          data[i] = Math.random() * 2 - 1;
        }
        const noise = this.ctx.createBufferSource();
        noise.buffer = buffer;
        const filter = this.ctx.createBiquadFilter();
        filter.type = 'lowpass';
        filter.frequency.setValueAtTime(800, this.ctx.currentTime);
        filter.frequency.exponentialRampToValueAtTime(30, this.ctx.currentTime + 1.5);
        const gain = this.ctx.createGain();
        gain.gain.setValueAtTime(1.0, this.ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 1.5);
        noise.connect(filter);
        filter.connect(gain);
        gain.connect(this.ctx.destination);
        noise.start();
      }
      boostPad() {
        if (!this.ctx) return;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(400, this.ctx.currentTime);
        osc.frequency.exponentialRampToValueAtTime(880, this.ctx.currentTime + 0.18);
        gain.gain.setValueAtTime(0.25, this.ctx.currentTime);
        gain.gain.linearRampToValueAtTime(0, this.ctx.currentTime + 0.18);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.18);
      }
      powerup() {
        if (!this.ctx) return;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'square';
        osc.frequency.setValueAtTime(260, this.ctx.currentTime);
        osc.frequency.linearRampToValueAtTime(600, this.ctx.currentTime + 0.2);
        gain.gain.setValueAtTime(0.2, this.ctx.currentTime);
        gain.gain.linearRampToValueAtTime(0, this.ctx.currentTime + 0.2);
        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start();
        osc.stop(this.ctx.currentTime + 0.2);
      }

    /* =========================================================================
       6. ARENA CONSTRUCTION (FUTURISTIC STADIUM)
       ========================================================================= */
    function createArena() {
      // Procedural Arena Floor Texture with Hex Grid and Field Lines
      const fCanvas = document.createElement('canvas');
      fCanvas.width = 1024;
      fCanvas.height = 2048;
      const ctx = fCanvas.getContext('2d');

      // Turf gradient
      const grad = ctx.createLinearGradient(0, 0, 0, 2048);
      grad.addColorStop(0, '#0a1d37');
      grad.addColorStop(0.5, '#0b1626');
      grad.addColorStop(1, '#3b180d');
      ctx.fillStyle = grad;
      ctx.fillRect(0, 0, 1024, 2048);

      // Subtle Tech Hex Pattern
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.05)';
      ctx.lineWidth = 1.5;
      const hexR = 30;
      for (let y = 0; y < 2048; y += hexR * 1.5) {
        for (let x = 0; x < 1024; x += hexR * 1.732) {
          ctx.beginPath();
          for (let a = 0; a < 6; a++) {
            const angle = (Math.PI / 3) * a;
            const hx = x + hexR * Math.cos(angle);
            const hy = y + hexR * Math.sin(angle);
            if (a === 0) ctx.moveTo(hx, hy);
            else ctx.lineTo(hx, hy);
          }
          ctx.closePath();
          ctx.stroke();
        }
      }

      // Field Lines
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.7)';
      ctx.lineWidth = 8;

      // Outer boundary
      ctx.strokeRect(40, 40, 1024 - 80, 2048 - 80);

      // Center Line
      ctx.beginPath();
      ctx.moveTo(40, 1024);
      ctx.lineTo(1024 - 40, 1024);
      ctx.stroke();

      // Center Circle
      ctx.beginPath();
      ctx.arc(512, 1024, 180, 0, Math.PI * 2);
      ctx.stroke();

      // Blue / Orange Goal Area Arcs
      ctx.strokeStyle = 'rgba(0, 210, 255, 0.8)';
      ctx.beginPath();
      ctx.arc(512, 40, 260, 0, Math.PI);
      ctx.stroke();

      ctx.strokeStyle = 'rgba(255, 110, 0, 0.8)';
      ctx.beginPath();

      const bodyGeo = new THREE.BoxGeometry(2.1, 0.75, 4.0);
      const bodyMat = new THREE.MeshStandardMaterial({
        color: isAI ? 0xe63946 : skinConfig.bodyColor,
        emissive: isAI ? 0x4a0007 : skinConfig.emissive,
        roughness: skinConfig.roughness,
        metalness: skinConfig.metalness
      });
      const bodyMesh = new THREE.Mesh(bodyGeo, bodyMat);
      bodyMesh.position.y = 0.55;
      bodyMesh.castShadow = true;
      bodyMesh.receiveShadow = true;
      carGroup.add(bodyMesh);

      // Cabin / Cockpit
      const cabinGeo = new THREE.BoxGeometry(1.6, 0.65, 2.1);
      const cabinMat = new THREE.MeshPhysicalMaterial({
        color: 0x111625,
        roughness: 0.1,
        metalness: 0.9,
        transmission: 0.3,
        transparent: true,
        opacity: 0.95
      });
      const cabinMesh = new THREE.Mesh(cabinGeo, cabinMat);
      cabinMesh.position.set(0, 1.05, -0.2);
      cabinMesh.castShadow = true;
      carGroup.add(cabinMesh);

      const wingMat = new THREE.MeshStandardMaterial({ color: 0x111111, roughness: 0.4 });
      const wingMesh = new THREE.Mesh(wingGeo, wingMat);
      wingMesh.position.set(0, 1.45, 1.6);

      carGroup.add(strutL);
      const strutR = strutL.clone();
      strutR.position.x = 0.7;
      carGroup.add(strutR);

      // Headlights
      const headMat = new THREE.MeshBasicMaterial({ color: isAI ? 0xffbb00 : 0x00f0ff });
      const headL = new THREE.Mesh(new THREE.BoxGeometry(0.4, 0.15, 0.1), headMat);
      headL.position.set(-0.75, 0.65, -2.01);
      carGroup.add(headL);
      const headR = headL.clone();
      headR.position.x = 0.75;
      carGroup.add(headR);

      // Twin Exhaust Nozzles
      const exhMat = new THREE.MeshStandardMaterial({ color: 0x444444, metalness: 0.9, roughness: 0.2 });
      const exhGeo = new THREE.CylinderGeometry(0.16, 0.16, 0.4, 12);
      const exhL = new THREE.Mesh(exhGeo, exhMat);
      exhL.rotation.x = Math.PI / 2;
      exhL.position.set(-0.5, 0.5, 2.02);
      carGroup.add(exhL);
      const exhR = exhL.clone();
      exhR.position.x = 0.5;

      const wheelOffsets = [
        { x: -1.15, y: 0.38, z: -1.25 },
        { x: 1.15, y: 0.38, z: -1.25 },
        { x: -1.15, y: 0.42, z: 1.25 },
        { x: 1.15, y: 0.42, z: 1.25 },

        } else {

          const rimGeo = new THREE.CylinderGeometry(tireRadius * 0.68, tireRadius * 0.68, 0.38, 8);
          const rimMat = new THREE.MeshStandardMaterial({ color: 0x999999, metalness: 0.8, roughness: 0.3 });
          const rim = new THREE.Mesh(rimGeo, rimMat);
          rim.rotation.z = Math.PI / 2;
          wGroup.add(rim);
        }

        wGroup.position.set(pos.x, pos.y, pos.z);
        carGroup.add(wGroup);
        wheels.push(wGroup);
      });

      return { mesh: carGroup, wheels };
    }

    class Car {
      constructor(isAI = false, startPos = new CANNON.Vec3(0, 1.5, 45), skinKey = 'neonBlue', wheelKey = 'classic') {
        this.isAI = isAI;
        this.boost = 33.3;
        this.isBoosting = false;
        this.grounded = false;
        this.jumpCount = 0;
        this.jumpTimer = 0;
        this.flipTorqueTime = 0;

        // Visual Construction
        const skin = isAI ? CAR_SKINS.crimsonRed : CAR_SKINS[skinKey];
        const carData = createCarMesh(skin, wheelKey, isAI);
        this.visual = carData.mesh;
        this.wheels = carData.wheels;
        scene.add(this.visual);

        // Physics Body (Low center of gravity for stable driving)
        const chassisShape = new CANNON.Box(new CANNON.Vec3(1.05, 0.45, 1.95));
        this.body = new CANNON.Body({
          mass: 220,
          material: carMaterial,
          linearDamping: 0.12,
          angularDamping: 0.45
        });
        this.body.addShape(chassisShape, new CANNON.Vec3(0, 0.45, 0));
        this.body.position.copy(startPos);
        if (isAI) {
          // AI faces player goal (-Z)
          this.body.quaternion.setFromEuler(0, 0, 0);
        } else {
          // Player faces Orange goal (+Z)
          this.body.quaternion.setFromEuler(0, Math.PI, 0);
        }
        world.addBody(this.body);

        // Flame Particles Emitter Pool
        this.flames = this.createFlameParticles();
        scene.add(this.flames.particles);

        // Rumble Active State
        this.rumblePower = null;
        this.rumbleTimer = 0;
        this.superSpeedActive = false;
      }

      createFlameParticles() {
        const count = 60;
        const geo = new THREE.BufferGeometry();
        const posArr = new Float32Array(count * 3);
        const colArr = new Float32Array(count * 3);
        const lifeArr = new Float32Array(count);

        for (let i = 0; i < count; i++) {
          posArr[i * 3 + 1] = -999;
          lifeArr[i] = 0;
          colArr[i * 3] = 1.0;
          colArr[i * 3 + 1] = 0.4 + Math.random() * 0.4;
          colArr[i * 3 + 2] = 0.05;
        }

        geo.setAttribute('position', new THREE.BufferAttribute(posArr, 3));
        geo.setAttribute('color', new THREE.BufferAttribute(colArr, 3));

        const mat = new THREE.PointsMaterial({
          size: 0.8,
          vertexColors: true,
          transparent: true,
          opacity: 0.85,
          blending: THREE.AdditiveBlending
        });

        const points = new THREE.Points(geo, mat);
        return { particles: points, count, life: lifeArr };
      }

      updateFlames(dt) {
        const pos = this.flames.particles.geometry.attributes.position.array;
        const life = this.flames.life;
        const isEmitting = this.isBoosting;

        const forward = new THREE.Vector3(0, 0, 1).applyQuaternion(this.visual.quaternion);
        const exhLeft = new THREE.Vector3(-0.5, 0.5, 2.0).applyQuaternion(this.visual.quaternion).add(this.visual.position);
        const exhRight = new THREE.Vector3(0.5, 0.5, 2.0).applyQuaternion(this.visual.quaternion).add(this.visual.position);

        for (let i = 0; i < this.flames.count; i++) {
          if (life[i] > 0) {
            life[i] -= dt * 3.5;
            pos[i * 3 + 0] += forward.x * dt * 8 + (Math.random() - 0.5) * 0.2;
            pos[i * 3 + 1] += forward.y * dt * 8 + (Math.random() - 0.5) * 0.2;
            pos[i * 3 + 2] += forward.z * dt * 8 + (Math.random() - 0.5) * 0.2;
            if (life[i] <= 0) pos[i * 3 + 1] = -999;
          } else if (isEmitting && Math.random() < 0.4) {
            life[i] = 1.0;
            const origin = Math.random() > 0.5 ? exhLeft : exhRight;
            pos[i * 3 + 0] = origin.x;
            pos[i * 3 + 1] = origin.y;
            pos[i * 3 + 2] = origin.z;
          }
        }
        this.flames.particles.geometry.attributes.position.needsUpdate = true;
      }

      checkGrounded() {
        // Cast check based on elevation and upright orientation
        const isUpright = this.body.quaternion.vmult(new CANNON.Vec3(0, 1, 0)).y > 0.35;
        const closeToGround = this.body.position.y < 1.15;
        this.grounded = closeToGround && isUpright;
        if (this.grounded) {
          this.jumpCount = 0;
        }
      }

      applyDriveControls(throttle, steer, boostKey, pitch, roll, jumpPressed, dt) {
        this.checkGrounded();

        // 1. Boost Execution
        const hasBoost = this.boost > 0 || this.superSpeedActive;
        this.isBoosting = boostKey && hasBoost;
        if (this.isBoosting) {
          if (!this.superSpeedActive) {
            this.boost = Math.max(0, this.boost - dt * 28);
          }
          const forward = this.body.quaternion.vmult(new CANNON.Vec3(0, 0, -1));
          const boostForce = this.superSpeedActive ? 6800 : 4400;
          this.body.applyForce(forward.scale(boostForce));
        }

        // 2. Grounded Driving Mechanics
        if (this.grounded) {
          const forward = this.body.quaternion.vmult(new CANNON.Vec3(0, 0, -1));
          const driveForce = 3500 * throttle;
          this.body.applyForce(forward.scale(driveForce));

          // Steering (Yaw)
          if (Math.abs(throttle) > 0.05 || Math.abs(this.body.velocity.length()) > 1.0) {
            const steerDir = throttle < 0 ? -steer : steer;
            this.body.angularVelocity.y = -steerDir * 2.8;
          }

          // Lateral Damping (High grip, no infinite side slipping)
          const right = this.body.quaternion.vmult(new CANNON.Vec3(1, 0, 0));
          const lateralSpeed = this.body.velocity.dot(right);
          this.body.velocity.vsub(right.scale(lateralSpeed * 0.15), this.body.velocity);
        } else {
          // 3. Mid-Air Aerial Flight Controls
          const airTorque = 3.6;
          // Pitch (W/S)
          if (Math.abs(pitch) > 0.1) {
            const pitchVec = this.body.quaternion.vmult(new CANNON.Vec3(1, 0, 0));
            this.body.angularVelocity.vadd(pitchVec.scale(pitch * airTorque * dt * 18), this.body.angularVelocity);
          }
          // Air Roll (Q/E)
          if (Math.abs(roll) > 0.1) {
            const rollVec = this.body.quaternion.vmult(new CANNON.Vec3(0, 0, 1));
            this.body.angularVelocity.vadd(rollVec.scale(-roll * airTorque * dt * 18), this.body.angularVelocity);
          }
          // Air Yaw (A/D)
          if (Math.abs(steer) > 0.1) {
            const yawVec = this.body.quaternion.vmult(new CANNON.Vec3(0, 1, 0));
            this.body.angularVelocity.vadd(yawVec.scale(-steer * airTorque * dt * 14), this.body.angularVelocity);
          }
        }

        // 4. Double Jump & Dodge Flips
        if (jumpPressed) {
          if (this.grounded && this.jumpCount === 0) {
            // First Jump off turf
            this.body.velocity.y = 11.5;
            this.jumpCount = 1;
            sfx.jump();
          } else if (this.jumpCount === 1) {
            // Second jump / Flip mechanic
            this.jumpCount = 2;
            sfx.dodge();
            if (Math.abs(pitch) > 0.3) {
              // Front or Back Dodge Flip
              const forward = this.body.quaternion.vmult(new CANNON.Vec3(0, 0, -1));
              const flipImpulse = forward.scale(pitch > 0 ? -12 : 12);
              this.body.velocity.vadd(flipImpulse, this.body.velocity);
              this.body.angularVelocity.x = pitch > 0 ? 12 : -12;
            } else if (Math.abs(steer) > 0.3) {
              // Side Roll Dodge
              const right = this.body.quaternion.vmult(new CANNON.Vec3(1, 0, 0));
              const dodgeImpulse = right.scale(steer * 11);
              this.body.velocity.vadd(dodgeImpulse, this.body.velocity);
              this.body.angularVelocity.z = steer * 12;
            } else {
              // Neutral Second Jump Pop
              this.body.velocity.y += 10.5;
            }
          }
        }

        // Sync visual mesh with Cannon physics body
        this.visual.position.copy(this.body.position);
        this.visual.quaternion.copy(this.body.quaternion);

        // Spin wheels when moving
        const speed = this.body.velocity.length();
        const spinSpeed = speed * dt * 3.5;
        this.wheels.forEach(w => {
          w.children[0].rotation.x += spinSpeed;
        });

        this.updateFlames(dt);
      }

      reset(x, z, faceZ) {
        this.body.position.set(x, 1.2, z);
        this.body.velocity.set(0, 0, 0);
        this.body.angularVelocity.set(0, 0, 0);
        this.body.quaternion.setFromEuler(0, faceZ > 0 ? 0 : Math.PI, 0);
        this.boost = 33.3;
        this.jumpCount = 0;
        this.visual.position.copy(this.body.position);
        this.visual.quaternion.copy(this.body.quaternion);
      }
    }

    function createSoccerBall() {
      // Textured Ball Mesh
      const bCanvas = document.createElement('canvas');
      bCanvas.width = 512;
      bCanvas.height = 256;
      const bctx = bCanvas.getContext('2d');
      bctx.fillStyle = '#f0f0f4';
      bctx.fillRect(0, 0, 512, 256);
      bctx.fillStyle = '#111827';
      // Hexagonal soccer markings
      for (let i = 0; i < 18; i++) {
        const cx = (i * 73) % 512;
        const cy = (i * 57) % 256;
        bctx.beginPath();
        bctx.arc(cx, cy, 22, 0, Math.PI * 2);
        bctx.fill();
      }
      const ballTex = new THREE.CanvasTexture(bCanvas);

      const ballGeo = new THREE.SphereGeometry(BALL_RADIUS, 32, 32);
      const ballMat = new THREE.MeshStandardMaterial({
        map: ballTex,
        roughness: 0.3,
        metalness: 0.15
      });
      const mesh = new THREE.Mesh(ballGeo, ballMat);
      mesh.castShadow = true;
      scene.add(mesh);

      // Physics Rigid Body
      const shape = new CANNON.Sphere(BALL_RADIUS);
      const body = new CANNON.Body({
        mass: 35,
        material: ballMaterial,
        linearDamping: 0.08,
        angularDamping: 0.15
      });
      body.addShape(shape);
      body.position.set(0, 6, 0);
      world.addBody(body);

      // Dynamic Height Shadow Disk on Pitch
      const shadowGeo = new THREE.RingGeometry(0, BALL_RADIUS * 1.3, 32);
      const shadowMat = new THREE.MeshBasicMaterial({
        color: 0x000000,
        transparent: true,
        opacity: 0.6,
        depthWrite: false
      });
      const shadowMesh = new THREE.Mesh(shadowGeo, shadowMat);
      shadowMesh.rotation.x = -Math.PI / 2;
      shadowMesh.position.y = 0.04;
      scene.add(shadowMesh);

      // Freeze Visual Ice Shell Mesh
      const iceGeo = new THREE.IcosahedronGeometry(BALL_RADIUS * 1.25, 2);
      const iceMat = new THREE.MeshPhysicalMaterial({
        color: 0x00ffff,
        transparent: true,
        opacity: 0.6,
        roughness: 0.1,
        transmission: 0.8
      });
      const iceMesh = new THREE.Mesh(iceGeo, iceMat);
      iceMesh.visible = false;
      scene.add(iceMesh);

      return { mesh, body, shadowMesh, iceMesh, isFrozen: false, freezeTimer: 0 };
    }

    /* =========================================================================
       11. RUMBLE POWER-UPS ENGINE
       ========================================================================= */
    const RUMBLE_TYPES = [
      { id: 'punch', name: 'Super Punch', icon: '🥊' },
      { id: 'magnet', name: 'Magnet', icon: '🧲' },
      { id: 'freeze', name: 'Freeze', icon: '❄️' },
      { id: 'speed', name: 'Super Speed', icon: '⚡' }

    let playerPowerUp = null;
    let powerUpChargeTimer = 15.0;

    // Visual Punch Glove Model
    const gloveGeo = new THREE.BoxGeometry(2, 2, 2.8);
    const gloveMat = new THREE.MeshStandardMaterial({ color: 0xff0044, metalness: 0.5, roughness: 0.2 });
    const punchGlove = new THREE.Mesh(gloveGeo, gloveMat);
    punchGlove.visible = false;

    function triggerRumble(powerId, userCar) {
      if (!powerId) return;
      sfx.powerup();

      if (powerId === 'punch') {
        // Launches boxing punch towards the ball
        punchGlove.visible = true;
        punchGlove.position.copy(userCar.visual.position);
        punchGlove.quaternion.copy(userCar.visual.quaternion);

        const toBall = ball.body.position.vsub(userCar.body.position);
        const dist = toBall.length();

        if (dist < 40) {
          toBall.normalize();
          ball.body.applyImpulse(toBall.scale(2200), ball.body.position);
          sfx.hit();
        }
        setTimeout(() => { punchGlove.visible = false; }, 400);

      } else if (powerId === 'magnet') {
        userCar.rumblePower = 'magnet';
        userCar.rumbleTimer = 4.0;

      } else if (powerId === 'freeze') {
        ball.isFrozen = true;
        ball.freezeTimer = 2.0;
        ball.body.velocity.set(0, 0, 0);
        ball.body.angularVelocity.set(0, 0, 0);
        ball.iceMesh.visible = true;

      } else if (powerId === 'speed') {
        userCar.superSpeedActive = true;
        userCar.rumblePower = 'speed';
        userCar.rumbleTimer = 5.0;
      }
    }

    function updateRumbleSystems(dt) {
      // 1. Charge Timer for Player
      if (!playerPowerUp) {
        powerUpChargeTimer -= dt;
        const iconEl = document.getElementById('powerup-icon');
        const nameEl = document.getElementById('powerup-name');

        if (powerUpChargeTimer <= 0) {
          const rand = RUMBLE_TYPES[Math.floor(Math.random() * RUMBLE_TYPES.length)];
          playerPowerUp = rand.id;
          iconEl.textContent = rand.icon;
          nameEl.textContent = `${rand.name} READY!`;
          nameEl.style.color = '#00ffaa';
        } else {
          iconEl.textContent = '⏳';
          nameEl.textContent = `Charging... (${Math.ceil(powerUpChargeTimer)}s)`;
          nameEl.style.color = '#8892b0';
        }
      }

      // 2. Active Rumble Power Durations
      [playerCar, aiCar].forEach(car => {
        if (!car) return;
        if (car.rumbleTimer > 0) {
          car.rumbleTimer -= dt;

          // Magnet Attraction Effect
          if (car.rumblePower === 'magnet') {
            const pullDir = car.body.position.vsub(ball.body.position);
            const dist = pullDir.length();
            if (dist < 45) {
              pullDir.normalize();
              const pullStrength = Math.min(1800, 35000 / (dist + 1));
              ball.body.applyForce(pullDir.scale(pullStrength));
            }
          }

          if (car.rumbleTimer <= 0) {
            car.superSpeedActive = false;
            car.rumblePower = null;
          }
        }
      });

      // 3. Freeze Ball Mechanic
      if (ball.isFrozen) {
        ball.freezeTimer -= dt;
        ball.body.velocity.set(0, 0, 0);
        ball.body.angularVelocity.set(0, 0, 0);
        ball.iceMesh.position.copy(ball.mesh.position);

        if (ball.freezeTimer <= 0) {
          ball.isFrozen = false;
          ball.iceMesh.visible = false;
        }
      }
    }

    /* =========================================================================
       12. GOAL EXPLOSION PARTICLES & EFFECTS

      // Scoreboard Update

      document.getElementById('score-orange').textContent = scoreOrange;
    }

    let matchTime = 180; // 3:00 Minutes

    window.addEventListener('keydown', (e) => {
      sfx.init();

      if (k === 'w' || k === 'arrowup') keys.w = true;
      if (k === 's' || k === 'arrowdown') keys.s = true;
      if (k === 'a' || k === 'arrowleft') keys.a = true;
      if (k === 'd' || k === 'arrowright') keys.d = true;

      if (k === 'q') keys.q = true;

      if (k === 'a' || k === 'arrowleft') keys.a = false;
      if (k === 'd' || k === 'arrowright') keys.d = false;
      if (k === 'shift') keys.shift = false;
      if (k === ' ') keys.space = false;
      if (k === 'q') keys.q = false;

    function resetKickoff() {
      playerCar.reset(0, 48, -1);
      aiCar.reset(0, -48, 1);

      ball.body.velocity.set(0, 0, 0);

    }

    // Smart AI Autonomous Agent
    function runAILogic(dt) {
      if (!aiCar || !ball) return;

      const aiPos = aiCar.body.position;
      const ballPos = ball.body.position;

      // Predict ball interception spot based on ball velocity
      const target = ballPos.vadd(ball.body.velocity.scale(0.35));

      // Calculate vector towards predicted target
      const toTarget = target.vsub(aiPos);
      const dist = toTarget.length();

      // Desired heading angle
      const targetAngle = Math.atan2(toTarget.x, -toTarget.z);

      // Current AI heading angle
      const aiEuler = new CANNON.Vec3();
      aiCar.body.quaternion.toEuler(aiEuler);
      let angleDiff = targetAngle - aiEuler.y;

      // Wrap angle between -PI and PI
      while (angleDiff > Math.PI) angleDiff -= Math.PI * 2;
      while (angleDiff < -Math.PI) angleDiff += Math.PI * 2;

      let throttle = 1.0;
      let steer = 0.0;
      let boostKey = false;
      let jumpKey = false;

      // Steer toward target
      if (angleDiff > 0.12) steer = 1.0;
      else if (angleDiff < -0.12) steer = -1.0;

      // Drive / Boost / Aerial contest
      if (Math.abs(angleDiff) < 0.4) {
        throttle = 1.0;
        // Boost if aligned and distant or shooting
        if (dist > 18 && aiCar.boost > 15) {
          boostKey = true;
        }
      } else if (Math.abs(angleDiff) > 2.0) {
        // Reverse if target is awkwardly behind
        throttle = -0.5;
        steer = -steer;
      }

      // Aerial Jump if ball is bouncing close overhead
      if (ballPos.y > 4.5 && dist < 14 && aiCar.grounded) {
        jumpKey = true;
      }

      aiCar.applyDriveControls(throttle, steer, boostKey, 0, 0, jumpKey, dt);
    }

    function startMatch() {
      scene.remove(previewGroup);

      playerCar = new Car(false, new CANNON.Vec3(0, 1.2, 48), selectedSkinKey, selectedWheelKey);
      aiCar = new Car(true, new CANNON.Vec3(0, 1.2, -48), 'crimsonRed', 'classic');

      resetKickoff();

    /* =========================================================================
       15. OPTIMIZED ANIMATION LOOP (LOCKED 60 FPS)
       ========================================================================= */
    const clock = new THREE.Clock();
    let accumulatedTime = 0;
    const fixedTimeStep = 1 / 60;

    function animate() {
      requestAnimationFrame(animate);

      let delta = clock.getDelta();
      delta = Math.min(delta, 0.1); // Prevent physics spiral on tab switch

      // Garage Menu Turntable Rotation
      if (!gameActive) {
        if (previewGroup) previewGroup.rotation.y += delta * 0.8;
        camera.position.set(0, 3.5, 7.5);
        camera.lookAt(0, 1.0, 0);
        renderer.render(scene, camera);
        return;
      }

      // Goal Slow-Mo Transition
      if (goalSlowMoTimer > 0) {
        goalSlowMoTimer -= delta;
        slowMoFactor = 0.25;
        // Expand Shockwave
        shockwaveMesh.scale.addScalar(delta * 40);
        shockwaveMat.opacity = Math.max(0, goalSlowMoTimer / 2.4);
        if (goalSlowMoTimer <= 0) {
          slowMoFactor = 1.0;
          resetKickoff();
        }
      }

      const simDelta = delta * slowMoFactor;

      // Cannon-es Physics Fixed Stepping
      world.step(fixedTimeStep, simDelta, 3);

      // 1. Controls & Player Updates
      const throttle = keys.w ? 1 : (keys.s ? -1 : 0);
      const steer = keys.d ? 1 : (keys.a ? -1 : 0);
      const pitch = keys.w ? 1 : (keys.s ? -1 : 0);
      const roll = keys.e ? 1 : (keys.q ? -1 : 0);

      playerCar.applyDriveControls(
        throttle,
        steer,
        keys.shift,
        pitch,
        roll,
        keys.jumpJustPressed,
        simDelta
      );
      keys.jumpJustPressed = false; // Consume single-frame jump event

      // 2. AI Opponent Execution
      runAILogic(simDelta);

      // 3. Rumble Power-Up Updates
      updateRumbleSystems(simDelta);

      // 4. Ball Mesh Sync & Dynamic Height Shadow
      ball.mesh.position.copy(ball.body.position);
      ball.mesh.quaternion.copy(ball.body.quaternion);

      // Project ground shadow directly beneath ball
      ball.shadowMesh.position.x = ball.body.position.x;
      ball.shadowMesh.position.z = ball.body.position.z;
      const height = Math.max(0.1, ball.body.position.y);
      const scale = Math.max(0.5, 1.0 - height / 35.0);
      ball.shadowMesh.scale.set(scale, scale, 1);
      ball.shadowMesh.material.opacity = Math.max(0.15, 0.7 - height / 30.0);

      // 5. Boost Pad Pickups & Animation
      boostPads.forEach(pad => {
        pad.ringMesh.rotation.z += delta * 2.0;
        if (pad.cooldown > 0) {
          pad.cooldown -= simDelta;
          if (pad.cooldown <= 0) {
            pad.mesh.material.emissiveIntensity = 1.0;
            pad.ringMesh.visible = true;
          }
        } else {
          // Check Player collection
          if (playerCar.body.position.distanceTo(pad.pos) < (pad.big ? 3.5 : 2.2)) {
            if (playerCar.boost < 100) {
              playerCar.boost = Math.min(100, playerCar.boost + pad.amount);
              pad.cooldown = pad.maxCooldown;
              pad.mesh.material.emissiveIntensity = 0.05;
              pad.ringMesh.visible = false;
              sfx.boostPad();
            }
          }
          // Check AI collection
          if (aiCar.body.position.distanceTo(pad.pos) < (pad.big ? 3.5 : 2.2)) {
            if (aiCar.boost < 100) {
              aiCar.boost = Math.min(100, aiCar.boost + pad.amount);
              pad.cooldown = pad.maxCooldown;
              pad.mesh.material.emissiveIntensity = 0.05;
              pad.ringMesh.visible = false;
            }
          }
        }
      });

      // 6. Goal Detection
      if (goalSlowMoTimer <= 0) {
        const bPos = ball.body.position;
        // Orange Goal at +ARENA_LENGTH/2
        if (bPos.z > ARENA_LENGTH / 2 && Math.abs(bPos.x) < GOAL_WIDTH / 2 && bPos.y < GOAL_HEIGHT) {
          triggerGoal('BLUE');
        }
        // Blue Goal at -ARENA_LENGTH/2
        else if (bPos.z < -ARENA_LENGTH / 2 && Math.abs(bPos.x) < GOAL_WIDTH / 2 && bPos.y < GOAL_HEIGHT) {
          triggerGoal('ORANGE');
        }
      }

      // 7. Match Timer Countdown
      if (matchTime > 0) {
        matchTime -= simDelta;
        const mins = Math.floor(Math.max(0, matchTime) / 60);
        const secs = Math.floor(Math.max(0, matchTime) % 60);
        document.getElementById('match-timer').textContent =
          `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
      }

      // 8. Boost Meter HUD Update
      const boostVal = Math.round(playerCar.boost);
      document.getElementById('boost-value').textContent = playerCar.superSpeedActive ? '∞' : boostVal;
      const strokeOffset = 345 - (345 * (playerCar.superSpeedActive ? 100 : boostVal)) / 100;
      document.getElementById('boost-fill').style.strokeDashoffset = strokeOffset;

      // 9. Camera Smoothing (Chase Cam vs. Ball Focus Cam)
      const carPos = playerCar.visual.position;
      const carForward = new THREE.Vector3(0, 0, 1).applyQuaternion(playerCar.visual.quaternion);

      if (cameraMode === 'ball') {
        // Ball Cam: Positions behind car, points right at ball
        const toBall = ball.mesh.position.clone().sub(carPos).normalize();
        const camTargetPos = carPos.clone().sub(toBall.multiplyScalar(13)).add(new THREE.Vector3(0, 5.2, 0));
        camera.position.lerp(camTargetPos, 0.12);
        camera.lookAt(ball.mesh.position.clone().add(new THREE.Vector3(0, 0.8, 0)));
      } else {
        // Standard Chase Cam: Locks directly behind vehicle
        const camTargetPos = carPos.clone().add(carForward.multiplyScalar(12)).add(new THREE.Vector3(0, 4.6, 0));
        camera.position.lerp(camTargetPos, 0.14);
        camera.lookAt(carPos.clone().add(new THREE.Vector3(0, 1.2, 0)));
      }

      renderer.render(scene, camera);
    }

    window.addEventListener('resize', () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    // Start 60 FPS Render Loop
    requestAnimationFrame(animate);
  </script>
